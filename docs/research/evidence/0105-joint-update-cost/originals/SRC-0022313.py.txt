reports = [BASE / '83_maintenance_scope_plan.md', BASE / '83_maintenance_findings.md', ROOT / 'output/research/TabICLv2_재설계_공동갱신비용_검토.md']
for p in reports:
    s = p.read_text(encoding='utf-8')
    assert '사례' not in s, str(p)
    for link in re.findall(r'\]\((C:/[^)]+)\)', s):
        assert Path(link).is_file(), link

manifest = json.loads((SRC / 'source_manifest.json').read_text(encoding='utf-8'))
pdf_total = 0
pdf_pages = 0
for entry in manifest:
    if 'sha256' in entry:
        p = SRC / entry['filename']
        assert sha256(p.read_bytes()).hexdigest() == entry['sha256']
        if p.suffix == '.pdf':
            pdf_total += p.stat().st_size
            pdf_pages += len(PdfReader(p).pages)
assert (pdf_total, pdf_pages) == (1851180, 12)

archive = ROOT / 'output/research/redesign_20260925_snapshot_25'
if archive.exists():
    raise RuntimeError('Snapshot25 already exists; refuse overwrite')
files = [ROOT/'AGENTS.md', *reports, BASE/'collect_maintenance_83.py', Path(__file__).resolve(), BASE/'progress.md', BASE/'candidate_decision_index.md', LEDGER, BASE/'rctl_torch.py', *sorted(SRC.iterdir())]
copied = []
for src in files:
    if not src.is_file():
        continue
    rel = src.relative_to(ROOT)
    dst = archive / rel
    dst.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(src, dst)
    digest = sha256(src.read_bytes()).hexdigest()
    assert sha256(dst.read_bytes()).hexdigest() == digest
    copied.append({'path': rel.as_posix(), 'bytes': src.stat().st_size, 'sha256': digest})
verification = {
    'verified_utc': datetime.now(timezone.utc).isoformat(),
    'snapshot': str(archive), 'copied_files': len(copied),
    'copied_bytes': sum(x['bytes'] for x in copied),
    'PDF_count': 2, 'PDF_pages': pdf_pages, 'PDF_bytes': pdf_total,
    'source_hashes_verified': True, 'copy_hashes_verified': True,
    'local_report_links_verified': True, 'terminology_verified': True,
    'cost_ledger_unchanged': True, 'model_experiments_executed': 0,
    'goal_complete': False, 'files': copied
}
(archive/'verification.json').write_text(json.dumps(verification, ensure_ascii=False, indent=2), encoding='utf-8')
(BASE/'83_archive_verification.json').write_text(json.dumps(verification, ensure_ascii=False, indent=2), encoding='utf-8')
