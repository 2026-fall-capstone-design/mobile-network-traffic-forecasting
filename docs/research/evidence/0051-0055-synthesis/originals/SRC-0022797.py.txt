records=[]
def fetch(name,url,cap=5_000_000):
    rec={'file':name,'url':url,'cap':cap,'access_date':'2026-09-26'}
    data=None
    try:
        with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Research source audit/1.0'}),timeout=25) as response:
            data=response.read(cap+1); rec.update(status=response.status,final_url=response.url)
        if len(data)>cap: raise ValueError('Size cap exceeded')
        if name.endswith('.pdf') and not data.startswith(b'%PDF-'): raise ValueError('Not PDF')
        path=out/name;path.write_bytes(data)
        rec.update(bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
        if name.endswith('.pdf'):
            reader=PdfReader(path);rec['pages']=len(reader.pages)
            path.with_suffix('.txt').write_text('\n\n'.join(f'--- PAGE {i+1} ---\n{page.extract_text()}' for i,page in enumerate(reader.pages)),encoding='utf-8')
    except Exception as exc:
        rec['error']=str(exc);data=None
    records.append(rec)
    log.write_text(json.dumps(records,ensure_ascii=False,indent=2),encoding='utf-8')
    print(json.dumps(rec,ensure_ascii=False),flush=True)
    return data
