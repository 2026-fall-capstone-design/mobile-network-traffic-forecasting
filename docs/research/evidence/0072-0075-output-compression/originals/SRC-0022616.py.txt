from pathlib import Path
import urllib.request, json, hashlib
from pypdf import PdfReader

base=Path(__file__).resolve().parent/'sources/model_mechanism_72'
base.mkdir(parents=True,exist_ok=True)
manifest=base/'manifest.json'
assert not manifest.exists(), 'Already collected; inspect existing files.'
items=[
 ('Globalization_abs.html','https://arxiv.org/abs/2507.11729'),
 ('Globalization_v1.html','https://arxiv.org/html/2507.11729v1'),
 ('Globalization_2507_11729v1.pdf','https://arxiv.org/pdf/2507.11729v1'),
 ('ForecastClustering_arxiv_abs.html','https://arxiv.org/abs/2305.00473'),
 ('ForecastClustering_KBS_metadata.json','https://api.crossref.org/works/10.1016/j.knosys.2025.113649')]
records=[]
pdf_bytes=0
for name,url in items:
    r={'name':name,'url':url,'access_date':'2026-09-26'}
    try:
        with urllib.request.urlopen(url,timeout=25) as resp:
            data=resp.read(20_000_001)
            r.update(status=resp.status,content_type=resp.headers.get('Content-Type'))
        assert len(data)<=20_000_000
        path=base/name
        assert not path.exists()
        if name.endswith('.pdf'):
            assert data.startswith(b'%PDF-')
            assert pdf_bytes+len(data)<=20_000_000
            pdf_bytes+=len(data)
        path.write_bytes(data)
        r.update(saved=True,bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
        if name.endswith('.pdf'):
            reader=PdfReader(path)
            pages=[p.extract_text() or '' for p in reader.pages]
            path.with_suffix('.txt').write_text('\n'.join(f'\n=== PDF PAGE {i+1} ===\n{t}' for i,t in enumerate(pages)),encoding='utf-8')
            r['pages']=len(pages)
            r['first_page_excerpt']=pages[0][:400]
    except Exception as exc:
        r.update(saved=False,error=f'{type(exc).__name__}: {exc}')
    records.append(r)
    manifest.write_text(json.dumps(records,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    print(json.dumps(r,ensure_ascii=False),flush=True)
