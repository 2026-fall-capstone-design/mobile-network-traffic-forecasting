from pathlib import Path
import urllib.request,json,hashlib
base=Path(__file__).resolve().parent/'sources/output_compression_73'
records=[]
def get(url,name,limit=3_000_000):
    rec={'url':url,'file':name}
    try:
        with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Research literature verification'}),timeout=25) as r:
            b=r.read(limit+1)
        assert len(b)<=limit
        (base/name).write_bytes(b)
        rec.update(bytes=len(b),sha256=hashlib.sha256(b).hexdigest(),saved=True)
        return b
    except Exception as e:rec.update(saved=False,error=str(e));return None
    finally:
        records.append(rec)
        (base/'followup_manifest.json').write_text(json.dumps(records,indent=2),encoding='utf-8')
b=get('https://api.github.com/repos/NGMLGroup/Time-Series-Clustering-with-GNNs/commits/main','GNN_commit.json')
if b:
    sha=json.loads(b)['sha']
    for name in ['model.py','predictor.py','pooling_functions.py']:
        get('https://raw.githubusercontent.com/NGMLGroup/Time-Series-Clustering-with-GNNs/'+sha+'/source/modules/'+name,'GNN_'+name)
get('https://cran.r-project.org/web/packages/mbrdr/mbrdr.pdf','MBRDR_CRAN_manual.pdf')
print(json.dumps(records,indent=2))
