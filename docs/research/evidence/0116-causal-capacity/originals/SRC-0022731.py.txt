from pathlib import Path
import datetime
import hashlib
import json
import urllib.request
from pypdf import PdfReader

DEST = Path(__file__).resolve().parent / 'sources/causal_capacity_87'
DEST.mkdir(parents=True, exist_ok=True)
url = 'https://www.ijcai.org/proceedings/2025/0627.pdf'
path = DEST / 'CIB_IJCAI2025.pdf'
assert not path.exists(), 'Do not repeat a completed download.'
record = {'url': url, 'accessed_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'discovered_via': 'Browser Google query: "time series forecasting" "bottleneck" "rank"; official IJCAI PDF URL visible in scholarly search result.', 'cap_bytes': 5_000_000}
try:
    req = urllib.request.Request(url, headers={'User-Agent': 'ResearchSourceReview/1.0'})
    with urllib.request.urlopen(req, timeout=20) as response:
        data = response.read(5_000_001)
        assert len(data) <= 5_000_000 and data.startswith(b'%PDF')
        path.write_bytes(data)
        record.update(status=response.status, final_url=response.url, bytes=len(data), sha256=hashlib.sha256(data).hexdigest())
    paper = PdfReader(path)
    texts = [f'\n--- PDF page {i+1} ---\n' + (page.extract_text() or '') for i, page in enumerate(paper.pages)]
    path.with_suffix('.txt').write_text('\n'.join(texts), encoding='utf-8')
    record['pages'] = len(paper.pages)
    print(json.dumps({'pages': len(paper.pages), 'bytes': len(data), 'first_page': texts[0][:3500]}, ensure_ascii=False))
except Exception as exc:
    record['error'] = type(exc).__name__ + ': ' + str(exc)
    print(json.dumps({'error': record['error']}))
(DEST / 'cib_source_manifest.json').write_text(json.dumps(record, ensure_ascii=False, indent=2), encoding='utf-8')
