def write_report(pages):
    lines = []
    for page in pages:
        for block in page:
            kind, *args = block
            if kind == 'table':
                headers, rows, widths = args
                lines += ['| ' + ' | '.join(headers) + ' |', '| ' + ' | '.join(['---'] * len(headers)) + ' |']
                lines += ['| ' + ' | '.join(row) + ' |' for row in rows]
            elif kind == 'ref':
                label, target = args
                target = target.replace('\\', '/')
                lines.append(f'- [{label}](<{target}>)')
            else:
                prefix = {'title': '# ', 'heading': '## ', 'sub': '### '}.get(kind, '')
                lines.append(prefix + args[0])
            lines.append('')
    REPORT.parent.mkdir(exist_ok=True, parents=True)
    REPORT.write_text('\n'.join(lines), encoding='utf-8')
    assert '\uc0ac\ub840' not in REPORT.read_text(encoding='utf-8')


def write_docx(pages):
    from docx import Document
    from docx.shared import Cm, Pt, RGBColor
    from docx.oxml import OxmlElement
    from docx.oxml.ns import qn
    from docx.enum.text import WD_ALIGN_PARAGRAPH
    from docx.enum.table import WD_TABLE_ALIGNMENT, WD_CELL_VERTICAL_ALIGNMENT
    from docx.opc.constants import RELATIONSHIP_TYPE as RT

    doc = Document()
    section = doc.sections[0]
    section.page_width, section.page_height = Cm(21), Cm(29.7)
    section.top_margin, section.bottom_margin = Cm(1.8), Cm(1.8)
    section.left_margin, section.right_margin = Cm(2.1), Cm(2.1)
    section.header_distance, section.footer_distance = Cm(.7), Cm(.8)
    for name, size in [('Normal', 10.5), ('Title', 20), ('Heading 1', 14), ('Heading 2', 11.5)]:
        s = doc.styles[name]
        s.font.name = '맑은 고딕'
        s.font.size = Pt(size)
        s.font.color.rgb = RGBColor(0, 0, 0)
        s.element.get_or_add_rPr().rFonts.set(qn('w:eastAsia'), '맑은 고딕')
        pf = s.paragraph_format
        pf.line_spacing = Pt(15 if name == 'Normal' else size * 1.3)
        pf.space_after = Pt(7)
        pf.widow_control = True
        if name != 'Normal':
            s.font.bold = True
            pf.keep_with_next = True
            pf.space_before = Pt(10 if name != 'Title' else 0)
        ppr = s.element.get_or_add_pPr()
        snap = OxmlElement('w:snapToGrid')
        snap.set(qn('w:val'), '0')
        ppr.append(snap)
        for border in list(ppr.findall(qn('w:pBdr'))):
            ppr.remove(border)
    doc.core_properties.title = 'TabICLv2 기반 clustering 연구의 근거 검토'
    doc.core_properties.subject = '확인된 예측 근거와 미확정 clustering 판단을 구분한 중간 검토'
    doc.core_properties.author = '연구 검토 자료'
    footer = section.footer.paragraphs[0]
    footer.alignment = WD_ALIGN_PARAGRAPH.RIGHT
    footer.add_run('중간 검토본  ·  ').font.size = Pt(9)
    field = OxmlElement('w:fldSimple')
    field.set(qn('w:instr'), 'PAGE')
    footer._p.append(field)

    def link(paragraph, text, url):
        if not url.startswith('http'):
            path = Path(url)
            assert path.exists(), path
            url = path.as_uri()
        relationship = paragraph.part.relate_to(url, RT.HYPERLINK, is_external=True)
        el = OxmlElement('w:hyperlink')
        el.set(qn('r:id'), relationship)
        run = OxmlElement('w:r')
        props = OxmlElement('w:rPr')
        fonts = OxmlElement('w:rFonts')
        for attr in ('ascii', 'hAnsi', 'eastAsia'):
            fonts.set(qn('w:' + attr), '맑은 고딕')
        props.append(fonts)
        color = OxmlElement('w:color'); color.set(qn('w:val'), '234E70'); props.append(color)
        size = OxmlElement('w:sz'); size.set(qn('w:val'), '18'); props.append(size)
        run.append(props)
        txt = OxmlElement('w:t'); txt.text = text; run.append(txt)
        el.append(run); paragraph._p.append(el)

    def table(headers, rows, widths):
        tbl = doc.add_table(rows=1, cols=len(headers))
        tbl.alignment = WD_TABLE_ALIGNMENT.CENTER
        tbl.autofit = False
        for col, width in zip(tbl.columns, widths):
            col.width = Cm(width)
        props = tbl._tbl.tblPr
        borders = OxmlElement('w:tblBorders')
        for edge in ('top', 'left', 'bottom', 'right', 'insideH', 'insideV'):
            el = OxmlElement('w:' + edge)
            for k, v in [('val', 'single'), ('sz', '4'), ('color', 'D9D9D9')]:
                el.set(qn('w:' + k), v)
            borders.append(el)
        props.append(borders)
        repeat = OxmlElement('w:tblHeader')
        tbl.rows[0]._tr.get_or_add_trPr().append(repeat)
        allrows = [headers] + rows
        for i, vals in enumerate(allrows):
            row = tbl.rows[0] if i == 0 else tbl.add_row()
            row._tr.get_or_add_trPr().append(OxmlElement('w:cantSplit'))
            for j, val in enumerate(vals):
                cell = row.cells[j]
                cell.width = Cm(widths[j])
                cell.vertical_alignment = WD_CELL_VERTICAL_ALIGNMENT.CENTER
                tcpr = cell._tc.get_or_add_tcPr()
                margins = OxmlElement('w:tcMar')
                for edge, twips in [('top','85'),('bottom','85'),('left','95'),('right','95')]:
                    el = OxmlElement('w:' + edge)
                    el.set(qn('w:w'), twips); el.set(qn('w:type'), 'dxa'); margins.append(el)
                tcpr.append(margins)
                if i == 0:
                    shade = OxmlElement('w:shd'); shade.set(qn('w:fill'), 'F2F2F2'); tcpr.append(shade)
                pp = cell.paragraphs[0]
                pp.paragraph_format.space_after = Pt(0)
                pp.paragraph_format.line_spacing = Pt(13)
                if i == 0:
                    pp.paragraph_format.keep_with_next = True
                rr = pp.add_run(val)
                rr.font.size = Pt(9.5)
                rr.bold = i == 0
        spacer = doc.add_paragraph()
        spacer.paragraph_format.space_after = Pt(0)
        spacer.paragraph_format.line_spacing = Pt(3)
        spacer.add_run().font.size = Pt(1)

    for page_index, page in enumerate(pages):
        for block_index, block in enumerate(page):
            kind, *args = block
            if kind == 'table':
                table(*args)
            elif kind == 'ref':
                pp = doc.add_paragraph()
                pp.paragraph_format.space_after = Pt(5)
                pp.paragraph_format.line_spacing = Pt(12.5)
                link(pp, args[0], args[1])
            else:
                style = {'title': 'Title', 'heading': 'Heading 1', 'sub': 'Heading 2'}.get(kind)
                pp = doc.add_paragraph(args[0], style=style)
                if page_index and block_index == 0:
                    pp.paragraph_format.page_break_before = True
                if kind in ('meta', 'note'):
                    for rr in pp.runs:
                        rr.font.size = Pt(9)
                    pp.paragraph_format.line_spacing = Pt(12.5)
                    pp.paragraph_format.space_after = Pt(6)
    DOCX.parent.mkdir(exist_ok=True, parents=True)
    doc.save(DOCX)
