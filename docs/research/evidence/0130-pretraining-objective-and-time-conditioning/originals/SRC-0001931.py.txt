"""Fetch selected official code at recorded commits, without importing it."""
from pathlib import Path
import hashlib
import json
import urllib.request
from datetime import datetime, timezone

ROOT = Path(__file__).resolve().parent
OUT = ROOT / 'sources' / 'pretraining_objective_100'
manifest = json.loads((OUT / 'manifest.json').read_text(encoding='utf-8'))
selection = {
    'TabClustPFN': ['model/tabcluster.py', 'exp/run_Softari.py', 'configs/train_config.py'],
    'APT': ['baselines/Normalization/normalization/APT.py', 'baselines/Normalization/runner/normal_tsf_runner.py', 'baselines/Normalization/ETTh1_itransformer.py', 'baselines/Normalization/loss/L2affine.py'],
}
for label, paths in selection.items():
    meta = json.loads((OUT / label / 'repository_metadata.json').read_text(encoding='utf-8'))
    for relative in paths:
        assert relative in meta['paths'], relative
        url = f"https://raw.githubusercontent.com/{meta['repo']}/{meta['commit']}/{relative}"
        req = urllib.request.Request(url, headers={'User-Agent': 'Research-source-verification/1.0'})
        with urllib.request.urlopen(req, timeout=30) as response:
            data = response.read(750_001)
        assert len(data) <= 750_000
        target = OUT / label / 'code' / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        if target.exists() and target.read_bytes() != data:
            raise FileExistsError(target)
        target.write_bytes(data)
        manifest['files'].append({'path': str(target.relative_to(ROOT)), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest(), 'source_url': url, 'kind': 'official_source_code_read_only'})
        manifest['network_bytes'] += len(data)
        assert manifest['network_bytes'] <= 15_000_000
        print(label, relative, len(data))
manifest['code_collected_utc'] = datetime.now(timezone.utc).isoformat()
(OUT / 'manifest.json').write_text(json.dumps(manifest, indent=2, ensure_ascii=False), encoding='utf-8')
