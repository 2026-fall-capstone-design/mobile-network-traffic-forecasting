"""Read-only source collection. Never imports downloaded code or any model."""
from pathlib import Path
import hashlib
import json
import shutil
import urllib.request
from datetime import datetime, timezone
from pypdf import PdfReader

ROOT = Path(__file__).resolve().parent
OUT = ROOT / 'sources' / 'pretraining_objective_100'
OUT.mkdir(parents=True, exist_ok=True)
manifest = []
network_bytes = 0

def fetch(url, limit=8_000_000):
    global network_bytes
    req = urllib.request.Request(url, headers={'User-Agent': 'Research-source-verification/1.0', 'Accept': 'application/pdf,application/json,text/plain,text/html,*/*'})
    with urllib.request.urlopen(req, timeout=40) as response:
        data = response.read(limit + 1)
        final_url = response.url
    if len(data) > limit:
        raise ValueError('Per-file size limit exceeded')
    network_bytes += len(data)
    if network_bytes > 15_000_000:
        raise ValueError('Total transfer size limit exceeded')
    return data, final_url

def save(relative, data, url=None, extra=None):
    target = OUT / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    if target.exists() and target.read_bytes() != data:
        raise FileExistsError(target)
    target.write_bytes(data)
    entry = {'path': str(target.relative_to(ROOT)), 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest(), 'source_url': url}
    if extra:
        entry.update(extra)
    manifest.append(entry)
    return target

old = ROOT / 'sources' / 'TabClustPFN_2026.pdf'
reader = PdfReader(old)
alltext = '\n'.join(page.extract_text() or '' for page in reader.pages)
if '2601.21656v3' not in alltext:
    raise ValueError('Existing TabClustPFN PDF version requires manual check')
papers = [
    ('TabClustPFN_2601.21656v3.pdf', old.read_bytes(), 'https://arxiv.org/pdf/2601.21656v3', {'reused_from': str(old.relative_to(ROOT))}),
]
apt_url = 'https://ojs.aaai.org/index.php/AAAI/article/download/38542/42504'
apt_data, apt_final = fetch(apt_url)
if not apt_data.startswith(b'%PDF'):
    raise ValueError('APT download is not a PDF')
papers.append(('APT_AAAI_2026.pdf', apt_data, apt_url, {'final_transport_url': apt_final, 'doi': '10.1609/aaai.v40i18.38542'}))
for name, data, url, extra in papers:
    path = save(name, data, url, extra)
    pdf = PdfReader(path)
    manifest[-1]['pages'] = len(pdf.pages)
    page_texts = [page.extract_text() or '' for page in pdf.pages]
    save(Path(name).with_suffix('.txt'), ('\n'.join(f'\n=== PAGE {i+1} ===\n{t}' for i, t in enumerate(page_texts))).encode('utf-8'), url, {'kind': 'extracted_text'})

for repo, label in [('Tianqi-Zhao/TabClustPFN', 'TabClustPFN'), ('blisky-li/APT', 'APT')]:
    meta_data, _ = fetch(f'https://api.github.com/repos/{repo}', 1_000_000)
    meta = json.loads(meta_data)
    branch = meta['default_branch']
    head_data, _ = fetch(f'https://api.github.com/repos/{repo}/commits/{branch}', 1_000_000)
    head = json.loads(head_data)
    sha = head['sha']
    tree_data, _ = fetch(f'https://api.github.com/repos/{repo}/git/trees/{sha}?recursive=1', 2_000_000)
    tree = json.loads(tree_data)
    paths = [item['path'] for item in tree['tree'] if item['type'] == 'blob']
    metadata = {'repo': repo, 'branch': branch, 'commit': sha, 'commit_date': head['commit']['committer']['date'], 'tree_truncated': tree.get('truncated', False), 'paths': paths}
    save(f'{label}/repository_metadata.json', json.dumps(metadata, indent=2, ensure_ascii=False).encode(), f'https://github.com/{repo}/tree/{sha}')
    initial = ['README.md', 'cluster.py'] if label == 'TabClustPFN' else ['README.md', 'trainnormal.py']
    for rel in initial:
        url = f'https://raw.githubusercontent.com/{repo}/{sha}/{rel}'
        data, _ = fetch(url, 600_000)
        save(f'{label}/code/{rel}', data, url)
    print(json.dumps({'repo': repo, 'commit': sha, 'relevant_paths': [p for p in paths if any(s in p.lower() for s in ('cluster', 'cardinality', 'prototype', 'affine', 'apt', 'normalization', 'normal_runner', 'config'))][:70]}, ensure_ascii=False))

(OUT / 'manifest.json').write_text(json.dumps({'retrieved_utc': datetime.now(timezone.utc).isoformat(), 'network_bytes': network_bytes, 'files': manifest}, indent=2, ensure_ascii=False), encoding='utf-8')
print(json.dumps({'files': len(manifest), 'network_bytes': network_bytes, 'pdf_pages': sum(e.get('pages', 0) for e in manifest)}, ensure_ascii=False))
