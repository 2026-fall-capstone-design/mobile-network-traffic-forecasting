from pathlib import Path
import urllib.request, json, hashlib
from pypdf import PdfReader
base=Path(__file__).resolve().parent/'sources/group_protection_66'
base.mkdir(exist_ok=True)
manifest=base/'manifest.json'
assert not manifest.exists(), 'Collection already completed.'
records=[]
for label,paper,version in [('MMR','2405.01709','v2'),('MRI','2602.04155','v2'),('Fair_network','2502.06743','v2')]:
    for suffix,url in [('abs.html',f'https://arxiv.org/abs/{paper}'),
                       (f'{paper}{version}.pdf',f'https://arxiv.org/pdf/{paper}')]:
        name=f'{label}_{suffix}'
        rec={'name':name,'url':url,'accessed_date':'2026-09-26'}
        try:
            with urllib.request.urlopen(url,timeout=30) as response:
                data=response.read(20_000_001)
                rec.update(status=response.status,resolved_url=response.url)
            assert len(data)<=20_000_000
            path=base/name
            assert not path.exists()
            if suffix.endswith('.pdf'): assert data.startswith(b'%PDF-')
            path.write_bytes(data)
            rec.update(saved=True,bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
            if suffix.endswith('.pdf'):
                reader=PdfReader(path)
                texts=[page.extract_text() for page in reader.pages]
                assert paper+version in texts[0], 'Version marker mismatch.'
                path.with_suffix('.txt').write_text('\n'.join(f'\n=== PDF PAGE {i+1} ===\n{t}' for i,t in enumerate(texts)),encoding='utf-8')
                rec.update(pages=len(reader.pages),version_marker=paper+version)
        except Exception as exc:
            rec.update(saved=False,error=f'{type(exc).__name__}: {exc}')
        records.append(rec)
manifest.write_text(json.dumps(records,indent=2),encoding='utf-8')
print(json.dumps(records,indent=2))
