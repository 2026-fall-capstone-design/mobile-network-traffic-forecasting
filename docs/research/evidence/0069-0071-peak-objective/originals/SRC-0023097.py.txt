from pathlib import Path
import subprocess,json,hashlib
b=Path(__file__).resolve().parent/'sources/peak_objective_69'
out=b/'renders';out.mkdir(exist_ok=True)
poppler=Path('C:/Users/jungj/.cache/codex-runtimes/codex-primary-runtime/dependencies/native/poppler/Library/bin/pdftoppm.exe')
records=[]
for stem,pages in [('Forecasters_Dilemma_Statistical_Science_2017',[3,6,7]),('DeepCog_JSAC_2020_author_version',[8,11,12])]:
    for p in pages:
        prefix=out/f'{stem}_p{p}'
        assert not prefix.with_suffix('.png').exists()
        subprocess.run([str(poppler),'-f',str(p),'-l',str(p),'-r','100','-png','-singlefile',str(b/(stem+'.pdf')),str(prefix)],check=True,capture_output=True)
        image=prefix.with_suffix('.png')
        records.append(dict(file=image.name,pdf=stem+'.pdf',page=p,sha256=hashlib.sha256(image.read_bytes()).hexdigest()))
(out/'manifest.json').write_text(json.dumps(records,indent=2),encoding='utf-8')
print(json.dumps(records,indent=2))
