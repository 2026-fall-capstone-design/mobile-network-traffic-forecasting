def balanced(D):
    gs=[[i] for i in range(16)]
    for _ in range(2):
        edges=sorted((float(D[np.ix_(a,b)].mean()),min(a),min(b),i,j) for i,a in enumerate(gs) for j,b in enumerate(gs) if i<j)
        used=set();new=[]
        for _,__,___,i,j in edges:
            if i in used or j in used:continue
            used.update([i,j]);new.append(sorted(gs[i]+gs[j]))
        assert len(used)==len(gs);gs=sorted(new,key=min)
    lab=np.zeros(16,int)
    for k,g in enumerate(gs):lab[g]=k
    assert np.array_equal(np.bincount(lab),[4]*4)
    return lab
def within(K,lab):
    total=float(np.trace(K))
    for c in np.unique(lab):
        ix=np.flatnonzero(lab==c);total-=float(K[np.ix_(ix,ix)].sum())/len(ix)
    return max(0.,total/16)
