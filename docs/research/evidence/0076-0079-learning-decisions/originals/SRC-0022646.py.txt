"""Static source snapshots; downloaded modules are never imported or executed."""
import hashlib, json, urllib.request
from pathlib import Path
from datetime import datetime, timezone

ROOT = Path(__file__).resolve().parent
OUT = ROOT / 'sources' / 'response_transfer_76'
COMMIT = json.loads((OUT / 'TabDistill_commit.json').read_text(encoding='utf-8-sig'))['sha']
FILES = ['readme.md', 'experiments/interaction_search/tabarena_single_mulindex.py',
         'experiments/TabDistill_downstream_comparison/compare_index_performance.py',
         'src/spectralexplain/interactions.py']
DEST = OUT / 'TabDistill_code'
records = []
for rel in FILES:
    url = f'https://raw.githubusercontent.com/Clouddelta/tab-distill/{COMMIT}/{rel}'
    dest = DEST / rel
    assert not dest.exists()
    req = urllib.request.Request(url, headers={'User-Agent': 'Research-source-verification/1.0'})
    with urllib.request.urlopen(req, timeout=30) as response:
        data = response.read(500001)
    assert len(data) <= 500000
    dest.parent.mkdir(parents=True, exist_ok=True)
    dest.write_bytes(data)
    records.append(dict(path=rel, url=url, bytes=len(data), sha256=hashlib.sha256(data).hexdigest(), accessed_utc=datetime.now(timezone.utc).isoformat()))
for rel in ['_sklearn/regressor.py', '_model/inference.py']:
    src = ROOT / 'runtime/Lib/site-packages/tabicl' / rel
    data = src.read_bytes()
    dest = OUT / 'installed_TabICL_2.2_code' / rel
    assert not dest.exists()
    dest.parent.mkdir(parents=True, exist_ok=True)
    dest.write_bytes(data)
    records.append(dict(path=str(dest.relative_to(OUT)), source=str(src), bytes=len(data), sha256=hashlib.sha256(data).hexdigest(), purpose='Static API review; no gradient/model execution'))
(OUT / 'code_manifest.json').write_text(json.dumps(dict(commit=COMMIT, records=records), ensure_ascii=False, indent=2), encoding='utf-8')
print(json.dumps(dict(commit=COMMIT, count=len(records), bytes=sum(r['bytes'] for r in records)), ensure_ascii=False))
