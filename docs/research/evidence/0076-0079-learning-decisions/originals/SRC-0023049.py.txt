"""Download selected pinned source files and render six paper pages. No model work."""
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor
from datetime import datetime, timezone
import urllib.request, hashlib, json, subprocess
BASE=Path(__file__).resolve().parent/'sources/conditional_membership_77'
CODE=BASE/'dynmix_code'
assert not (CODE/'selected_files_manifest.json').exists()
sha=json.loads((CODE/'commit.json').read_text(encoding='utf-8'))['sha']
def fetch(name):
 url=f'https://raw.githubusercontent.com/victhorio/dynmix/{sha}/{name}'
 req=urllib.request.Request(url,headers={'User-Agent':'Research-source-verification/1.0'})
 with urllib.request.urlopen(req,timeout=30) as response:data=response.read(300001)
 assert len(data)<=300000
 target=CODE/name;target.parent.mkdir(parents=True,exist_ok=True);assert not target.exists();target.write_bytes(data)
 return {'url':url,'path':str(target),'commit':sha,'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest(),'accessed_utc':datetime.now(timezone.utc).isoformat()}
with ThreadPoolExecutor(max_workers=4) as pool:records=list(pool.map(fetch,['README.md','dynmix/dynamic.py','dynmix/dirichlet.py','dynmix/dlm.py']))
(CODE/'selected_files_manifest.json').write_text(json.dumps(records,ensure_ascii=False,indent=2),encoding='utf-8')
exe=Path('C:/Users/jungj/.cache/codex-runtimes/codex-primary-runtime/dependencies/native/poppler/Library/bin/pdftoppm.exe')
render=BASE/'rendered';render.mkdir(exist_ok=True)
jobs=[('DynamicClustering_2002.01890v1',10),('DynamicClustering_2002.01890v1',11),('DynaSTar_IJCAI2026',3),('DynaSTar_IJCAI2026',7),('CellularLoadBalancing_InformationFusion2024',4),('CellularLoadBalancing_InformationFusion2024',9)]
def convert(job):
 name,page=job
 target=render/f'{name}_p{page}'
 assert not Path(str(target)+'.png').exists()
 subprocess.run([str(exe),'-f',str(page),'-l',str(page),'-scale-to','1500','-png','-singlefile',str(BASE/(name+'.pdf')),str(target)],check=True,timeout=30,capture_output=True)
 path=Path(str(target)+'.png')
 return {'source':name+'.pdf','page_1based':page,'path':str(path),'sha256':hashlib.sha256(path.read_bytes()).hexdigest()}
with ThreadPoolExecutor(max_workers=3) as pool:pages=list(pool.map(convert,jobs))
(BASE/'render_manifest.json').write_text(json.dumps(pages,ensure_ascii=False,indent=2),encoding='utf-8')
print(json.dumps({'code':len(records),'commit':sha,'rendered_pages':pages},ensure_ascii=False))
