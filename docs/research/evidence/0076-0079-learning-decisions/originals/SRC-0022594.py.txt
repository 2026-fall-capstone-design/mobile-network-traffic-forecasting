"""Bounded public-source collection; no installation, model calls, or imported external code."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, urllib.request
from concurrent.futures import ThreadPoolExecutor
from pypdf import PdfReader

OUT=Path(__file__).resolve().parent/'sources/conditional_membership_77'
assert not (OUT/'extra_manifest.json').exists()
items=[
 ('CellularLoadBalancing_InformationFusion2024.pdf','https://bimsa.net/doc/publication/3963.pdf',5000000),
 ('dynmix_pypi.json','https://pypi.org/pypi/dynmix/json',200000),
 ('dynmix_repo.json','https://api.github.com/repos/vsartor/dynmix',200000),
 ('dynastar_repo.json','https://api.github.com/repos/ShaunHao/dynastar',200000),
]
def get(item):
 name,url,limit=item
 rec={'name':name,'url':url,'accessed_utc':datetime.now(timezone.utc).isoformat()}
 try:
  request=urllib.request.Request(url,headers={'User-Agent':'Research-source-verification/1.0'})
  with urllib.request.urlopen(request,timeout=45) as res:
   data=res.read(limit+1)
   rec.update(final_url=res.url,content_type=res.headers.get('Content-Type'))
  assert len(data)<=limit
  if name.endswith('.pdf'): assert data.startswith(b'%PDF-')
  else: json.loads(data)
  target=OUT/name;assert not target.exists();target.write_bytes(data)
  rec.update(path=str(target),bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
  if name.endswith('.pdf'):
   pdf=PdfReader(target)
   rec['pages']=len(pdf.pages);rec['valid_pdf']=True
   target.with_suffix('.txt').write_text('\n\n'.join(f'=== PDF PAGE {j+1} ===\n{p.extract_text(extraction_mode="layout")}' for j,p in enumerate(pdf.pages)),encoding='utf-8')
 except Exception as e:rec['error']=repr(e)
 return rec
with ThreadPoolExecutor(max_workers=4) as pool: records=list(pool.map(get,items))
(OUT/'extra_manifest.json').write_text(json.dumps(records,ensure_ascii=False,indent=2),encoding='utf-8')
print(json.dumps(records,ensure_ascii=False))
