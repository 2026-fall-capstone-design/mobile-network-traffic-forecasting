from pathlib import Path
import json
base=Path(__file__).resolve().parent
p=base/'cumulative_execution_budget.json'
ledger=json.loads(p.read_text(encoding='utf-8'))
r=json.loads((base/'results/aggregation_recovery_49/result.json').read_text(encoding='utf-8'))
f=json.loads((base/'results/aggregation_recovery_49/run_finished.json').read_text(encoding='utf-8'))
key='aggregation_recovery_49'
assert key not in ledger['new_cheap_diagnostic_seconds']
assert f['hashes_unchanged'] and r['all_finite_checks_passed']
assert r['new_model_fits']==r['new_TabICL_contexts']==r['new_RCTL_calls']==0
ledger['new_cheap_diagnostic_seconds'][key]=r['elapsed_seconds']
ledger['as_of_date']='2026-09-26'
cheap=sum(ledger['new_cheap_diagnostic_seconds'].values())
total=cheap+sum(ledger['used'][k] for k in ['TabICL_fit_predict_seconds','RCTL_wall_seconds','RCTL_frozen_inference_seconds'])
assert cheap<ledger['caps']['cheap_numeric_seconds'] and total<ledger['caps']['total_local_modeling_seconds']
assert ledger['used']['RCTL_fits']==29 and ledger['remaining_count_budgets']['RCTL_fits']==0
p.write_text(json.dumps(ledger,indent=2,allow_nan=False,ensure_ascii=False)+'\n',encoding='utf-8')
print(json.dumps({'added':key,'seconds':r['elapsed_seconds'],'cheap_total':cheap,'recorded_modeling_seconds':total,'remaining':ledger['remaining_count_budgets']}))
