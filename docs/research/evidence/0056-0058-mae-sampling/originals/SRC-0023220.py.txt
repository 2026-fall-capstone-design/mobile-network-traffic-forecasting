from pathlib import Path
import json
base=Path(__file__).resolve().parent
out=base/'results/mae_sampling_57'
path=base/'cumulative_execution_budget.json'
ledger=json.loads(path.read_text(encoding='utf-8'))
result=json.loads((out/'result.json').read_text(encoding='utf-8'))
key='mae_sampling_audit_57'
assert key not in ledger['new_cheap_diagnostic_seconds']
backup=out/'budget_before_accounting.json';assert not backup.exists()
backup.write_bytes(path.read_bytes())
ledger['new_cheap_diagnostic_seconds'][key]=result['numeric_seconds']
assert ledger['used']['RCTL_fits']==29 and ledger['remaining_count_budgets']['RCTL_fits']==0
cheap=sum(ledger['new_cheap_diagnostic_seconds'].values())
total=cheap+sum(ledger['used'][k] for k in ['TabICL_fit_predict_seconds','RCTL_wall_seconds','RCTL_frozen_inference_seconds'])
assert cheap<ledger['caps']['cheap_numeric_seconds'] and total<ledger['caps']['total_local_modeling_seconds']
path.write_text(json.dumps(ledger,ensure_ascii=False,indent=2,allow_nan=False)+'\n',encoding='utf-8')
print(json.dumps({'added':key,'cheap_total_seconds':cheap,'recorded_modeling_seconds':total,'remaining':ledger['remaining_count_budgets']}))
