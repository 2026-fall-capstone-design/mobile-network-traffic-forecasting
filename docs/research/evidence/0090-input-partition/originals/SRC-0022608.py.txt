from pathlib import Path
from datetime import datetime, timezone
from html.parser import HTMLParser
import urllib.request, json, hashlib

BASE=Path(__file__).resolve().parent/'sources/input_partition_81'
BASE.mkdir(parents=True,exist_ok=True)
MANIFEST=BASE/'source_manifest.json'
assert not MANIFEST.exists(), 'Completed collection must not be rerun'
records=[]
class Parser(HTMLParser):
    def __init__(self): super().__init__(); self.parts=[]; self.skip=0; self.math=0
    def handle_starttag(self,t,a):
        if t in ('script','style','nav'): self.skip+=1
        if t=='math': self.math+=1; self.parts.append(' '+dict(a).get('alttext','[math]')+' ')
        if t in ('section','p','h1','h2','h3','h4','li','tr','figure'): self.parts.append('\n')
    def handle_endtag(self,t):
        if t in ('script','style','nav'): self.skip=max(0,self.skip-1)
        if t=='math': self.math=max(0,self.math-1)
        if t in ('p','section','li','tr'): self.parts.append('\n')
    def handle_data(self,d):
        if not self.skip and not self.math: self.parts.append(d)
def fetch(name,url,limit=4000000):
    r={'name':name,'url':url,'accessed_utc':datetime.now(timezone.utc).isoformat()}
    try:
        req=urllib.request.Request(url,headers={'User-Agent':'Research-source-verification/1.0'})
        with urllib.request.urlopen(req,timeout=20) as f:
            if int(f.headers.get('Content-Length') or 0)>limit: raise ValueError('size limit')
            data=f.read(limit+1); r['final_url']=f.url
        assert len(data)<=limit
        p=BASE/name; assert not p.exists(); p.write_bytes(data)
        r.update(bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
        if name.endswith('.html'):
            parser=Parser(); parser.feed(data.decode('utf-8'))
            lines=[' '.join(s.split()) for s in ''.join(parser.parts).splitlines()]
            p.with_suffix('.txt').write_text('\n'.join(s for s in lines if s),encoding='utf-8')
        return data
    except Exception as e: r['error']=repr(e)
    finally:
        records.append(r); MANIFEST.write_text(json.dumps(records,ensure_ascii=False,indent=2),encoding='utf-8')
        print(json.dumps(r,ensure_ascii=False),flush=True)

for name,identifier in [('DUET','2412.10859v3'),('DGCformer','2405.08440v1'),('CCM','2404.01340v2'),('Predictive_variable_clustering','2312.16544v1')]:
    fetch(name+'_'+identifier+'.html','https://arxiv.org/html/'+identifier)
    fetch(name+'_metadata.html','https://arxiv.org/abs/'+identifier)
for name,repo in [('duet','decisionintelligence/DUET'),('ccm','Graph-and-Geometric-Learning/TimeSeriesCCM')]:
    data=fetch(name+'_commit.json','https://api.github.com/repos/'+repo+'/commits/HEAD')
    if data:
        sha=json.loads(data)['sha']
        data=fetch(name+'_tree.json',f'https://api.github.com/repos/{repo}/git/trees/{sha}?recursive=1')
        if data:
            paths=[x['path'] for x in json.loads(data)['tree'] if x['type']=='blob' and x['path'].endswith('.py')]
            print(json.dumps({'repo':repo,'commit':sha,'python_paths':paths},ensure_ascii=False),flush=True)
