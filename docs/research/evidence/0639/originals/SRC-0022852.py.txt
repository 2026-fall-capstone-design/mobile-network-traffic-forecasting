# The verified306 training function is appended by the one-time build script.

def train():
    import numpy as np
    import psutil
    import torch
    from rctl_torch import RCTL
    assert not (O / 'run_started.json').exists(), 'Never rerun a started training process.'
    settings = load(O / 'frozen_settings.json')
    auth = load(O / 'budget_authorization.json')
    assert sha(O / 'frozen_settings.json') == auth['settings_sha256']
    assert sha(L) == auth['ledger_reserved_sha256']
    assert all(sha(B / p) == h for p, h in settings['source_hashes'].items())
    torch.set_num_threads(1)
    torch.set_num_interop_threads(2)
    proc = psutil.Process()
    available_cpus = proc.cpu_affinity()
    proc.cpu_affinity(available_cpus[:4])
    state = {'fits_started': 0, 'fits_completed': 0, 'epochs_started': 0,
             'epochs_completed': 0, 'optimizer_batches': 0, 'results': [],
             'active_task': None, 'peak_rss_bytes': 0}
    write(O / 'run_started.json', {'at': now(), 'pid': os.getpid(),
          'settings_sha256': auth['settings_sha256'], 'torch': torch.__version__,
          'numpy': np.__version__, 'math_threads': torch.get_num_threads(),
          'interop_threads': torch.get_num_interop_threads(), 'cpu_affinity': proc.cpu_affinity()})
    start = time.perf_counter()
    stop = threading.Event()

    def guard():
        while not stop.wait(.2):
            elapsed = time.perf_counter() - start
            state['peak_rss_bytes'] = max(state['peak_rss_bytes'], proc.memory_info().rss)
            if elapsed > settings['wall_cap_seconds'] or state['peak_rss_bytes'] > settings['ram_cap_bytes']:
                write(O / 'run_finished.json', {'at': now(), 'success': False,
                      'wall_seconds': elapsed, 'state': copy.deepcopy(state),
                      'reason': 'precommitted wall or RAM limit'})
                os._exit(72)
    threading.Thread(target=guard, daemon=True).start()

    def predict(model, x):
        model.eval()
        with torch.no_grad():
            return torch.cat([model(x[i:i+512]) for i in range(0, len(x), 512)]).cpu().numpy()

    try:
        with np.load(B / settings['prepared_arrays']) as d:
            seq, y, times, ids, scales = (d[k] for k in ['seq', 'y', 'times', 'cell_ids', 'scales'])
        assert np.array_equal(ids, settings['cell_ids'])
        assert np.isfinite(seq).all() and np.isfinite(y).all()
        tr, va, te = times < 840, (times >= 840) & (times < 1008), times >= 1008
        assert (tr.sum(), va.sum(), te.sum()) == (672, 168, 480)
        with np.load(B / 'results/rctl_pilot/evaluation_data.npz') as prior:
            for key, actual in [('cell_ids', ids), ('test_y', y[:, te]), ('scales', scales), ('test_times', times[te])]:
                assert np.array_equal(prior[key], actual), key
        np.savez_compressed(O / 'evaluation_data.npz', cell_ids=ids, scales=scales,
                            test_times=times[te], test_y=y[:, te])
        for task in settings['tasks']:
            method, seed, cluster, members = (task[k] for k in ['method', 'seed', 'cluster', 'members'])
            key = f'{method}_{seed}_cluster{cluster}'
            if task['reuse']:
                src = task['reuse']['record']['artifacts']['.npz']
                assert sha(B / src['relative_path']) == src['sha256']
                with np.load(B / src['relative_path']) as data:
                    assert np.array_equal(data['members'], members)
                    assert np.array_equal(data['cell_ids'], ids[members])
                    assert np.array_equal(data['y'], y[members][:, te])
                # Keep the original prediction bytes, with the original provenance.
                (O / (key + '.npz')).write_bytes((B / src['relative_path']).read_bytes())
                state['results'].append({'method': method, 'seed': seed, 'cluster': cluster,
                      'members': members, 'reused': True, 'source': src})
                write(O / 'fit_results.json', state['results'])
                continue
            assert state['fits_started'] < settings['max_new_fits']
            state['fits_started'] += 1
            state['active_task'] = task
            write(O / 'attempt_progress.json', state)
            random.seed(seed)
            np.random.seed(seed)
            torch.manual_seed(seed)
            gen = torch.Generator().manual_seed(seed)
            trainx = torch.from_numpy(seq[members][:, tr].reshape(-1, 8, 9))
            trainy = torch.from_numpy(y[members][:, tr].reshape(-1))
            valx = torch.from_numpy(seq[members][:, va].reshape(-1, 8, 9))
            valy = y[members][:, va].reshape(-1)
            testx = torch.from_numpy(seq[members][:, te].reshape(-1, 8, 9))
            model = RCTL(steps=8, channels=9)
            optimizer = torch.optim.Adam((p for p in model.parameters() if p.requires_grad), lr=.001, eps=1e-7)
            best, best_state, best_epoch, stale = float('inf'), None, 0, 0
            hist, fit_start, reason = [], time.perf_counter(), 'epoch_cap'
            print(json.dumps({'event': 'fit_start', 'method': method, 'cluster': cluster}), flush=True)
            for epoch in range(1, 81):
                state['epochs_started'] += 1
                assert state['epochs_started'] <= settings['max_new_epochs']
                write(O / 'attempt_progress.json', state)
                model.train()
                order = torch.randperm(len(trainx), generator=gen)
                total = 0.
                for i in range(0, len(order), 256):
                    ix = order[i:i+256]
                    optimizer.zero_grad(set_to_none=True)
                    loss = torch.abs(model(trainx[ix]) - trainy[ix]).mean()
                    if not torch.isfinite(loss):
                        raise RuntimeError('Nonfinite loss')
                    loss.backward()
                    optimizer.step()
                    state['optimizer_batches'] += 1
                    total += float(loss.detach()) * len(ix)
                val = float(np.abs(predict(model, valx) - valy).mean())
                if not np.isfinite(val):
                    raise RuntimeError('Nonfinite validation loss')
                state['epochs_completed'] += 1
                hist.append({'epoch': epoch, 'train_mae': total/len(trainx), 'val_mae': val,
                             'seconds_from_fit_start': time.perf_counter()-fit_start})
                if val < best:
                    best, best_epoch, best_state, stale = val, epoch, copy.deepcopy(model.state_dict()), 0
                else:
                    stale += 1
                payload = dict(task=task, epoch=epoch, next_epoch=epoch+1,
                    model_state=copy.deepcopy(model.state_dict()), optimizer_state=optimizer.state_dict(),
                    best_state=best_state, best=best, best_epoch=best_epoch, stale=stale, history=hist,
                    random_state=random.getstate(), numpy_state=np.random.get_state(),
                    torch_rng=torch.get_rng_state(), permutation_rng=gen.get_state(),
                    settings_sha256=auth['settings_sha256'], run_key=KEY)
                temporary = O / (key+'_resume.pending.pt')
                torch.save(payload,temporary)
                for retry in range(8):
                    try:
                        os.replace(temporary,O / (key+'_resume.pt'))
                        break
                    except PermissionError:
                        if retry==7:raise
                        time.sleep(.05*(retry+1))
                write(O / (key + '_history.json'), hist)
                write(O / 'attempt_progress.json', state)
                if stale >= 10:
                    reason = 'validation_patience'
                    break
            assert best_state is not None
            model.load_state_dict(best_state)
            pred = predict(model, testx).reshape(len(members), 480)
            assert np.isfinite(pred).all()
            np.savez_compressed(O / (key + '.npz'), prediction=pred, y=y[members][:, te],
                                cell_ids=ids[members], members=members)
            torch.save(best_state, O / (key + '.pt'))
            rec = {'method': method, 'seed': seed, 'cluster': cluster, 'members': members,
                   'reused': False, 'epochs_run': len(hist), 'best_epoch': best_epoch,
                   'best_val_mae': best, 'stop_reason': reason, 'fit_seconds': time.perf_counter()-fit_start}
            state['results'].append(rec)
            state['fits_completed'] += 1
            state['active_task'] = None
            write(O / 'fit_results.json', state['results'])
            write(O / 'attempt_progress.json', state)
            print(json.dumps({'event': 'fit_done', **rec}), flush=True)
        assert len(state['results']) == len(settings['tasks']) and state['fits_completed'] == settings['max_new_fits']
        write(O / 'run_finished.json', {'at': now(), 'success': True,
              'wall_seconds': time.perf_counter()-start, 'state': state})
    except BaseException as exc:
        write(O / 'run_finished.json', {'at': now(), 'success': False,
              'wall_seconds': time.perf_counter()-start, 'state': state, 'reason': repr(exc)})
        raise
    finally:
        stop.set()

if __name__ == '__main__':
    {'prepare':prepare,'reserve':reserve,'train':train}[sys.argv[1]]()
