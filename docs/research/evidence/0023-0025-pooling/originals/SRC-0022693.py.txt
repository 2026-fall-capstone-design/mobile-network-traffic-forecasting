def optimum(atoms,weights):
    # atoms [cell, query, atom], weights [cell, query], unnormalised group mass.
    n,q,b=atoms.shape
    values=atoms.transpose(1,0,2).reshape(q,n*b)
    w=np.repeat(weights.T/b,b,axis=1)
    order=np.argsort(values,axis=1,kind='stable')
    sorted_v=np.take_along_axis(values,order,axis=1)
    sorted_w=np.take_along_axis(w,order,axis=1)
    mass=weights.sum(0)
    ix=(np.cumsum(sorted_w,axis=1)>=mass[:,None]/2).argmax(1)
    action=sorted_v[np.arange(q),ix]
    cost=(np.abs(atoms-action[None,:,None]).mean(2)*weights).sum(0)
    return action,cost,mass

def excess(atoms,weights,groups):
    # Local median is an optimal absolute-loss action for each discrete approximation.
    local=np.abs(atoms-np.median(atoms,axis=2,keepdims=True)).mean(2)
    base_cost=(local*weights).sum(0)
    total=np.zeros(atoms.shape[1]); outputs={}; zero={}
    for k,g in enumerate(groups):
        a,c,m=optimum(atoms[g],weights[g])
        outputs[k]=a;zero[k]=m<=0;total+=c
    delta=total-base_cost
    assert delta.min()>-1e-8
    return np.maximum(delta,0),outputs,zero
