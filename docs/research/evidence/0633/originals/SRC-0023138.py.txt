def read(p):
    until=time.perf_counter()+1
    while True:
        try:return json.loads(p.read_text(encoding='utf-8-sig'))
        except PermissionError:
            if time.perf_counter()>until:raise
            time.sleep(.005)

def run():
    import psutil
    from run_joint_ID_decision_558 import terminate_tree
    cfg,a=read(O/'frozen_settings.json'),read(O/'authorization.json')
    assert sha(L)==a['ledger_reserved_sha256'] and sha(O/'frozen_settings.json')==a['settings_sha256']
    assert all(sha(B/p)==h for p,h in cfg['source_hashes'].items()) and not (O/'run_started.json').exists()
    meter={k:0 for name in ['contexts','query_rows'] for k in [name+'_attempted',name+'_completed']}
    meter.update(Tab_seconds=0,phase='import',phase_started_perf=None);write(O/'meter.json',meter)
    start=time.perf_counter();proc=None;peak=0;failure=None;success=False;code=None
    write(O/'run_started.json',dict(at=now(),controller_pid=os.getpid()))
    try:
        with (O/'stdout.log').open('w',encoding='utf-8') as out,(O/'stderr.log').open('w',encoding='utf-8') as err:
            proc=subprocess.Popen([sys.executable,'-X','utf8',str(Path(__file__).resolve()),'worker'],stdout=out,stderr=err,creationflags=getattr(subprocess,'CREATE_NO_WINDOW',0))
            write(O/'worker_handle.json',dict(pid=proc.pid,create_time=psutil.Process(proc.pid).create_time()))
            while proc.poll() is None:
                meter=read(O/'meter.json');elapsed=time.perf_counter()-start
                pending=max(0,time.perf_counter()-meter['phase_started_perf']) if meter['phase'].startswith('Tab_') and meter['phase_started_perf'] is not None else 0
                tab=meter['Tab_seconds']+pending
                try:
                    pp=psutil.Process(proc.pid);peak=max(peak,pp.memory_info().rss+sum(ch.memory_info().rss for ch in pp.children(recursive=True)))
                except psutil.NoSuchProcess:pass
                # Stop with a small margin, preserving time for process reaping.
                if tab>LIMITS['Tab_seconds']-.5 or elapsed-tab>LIMITS['other_seconds']-.5 or elapsed>LIMITS['stage_seconds']-.5 or peak>LIMITS['rss_bytes']:
                    failure=dict(reason='resource_cap',Tab_seconds=tab,other_seconds=elapsed-tab,RSS=peak);break
                time.sleep(.1)
            if failure:terminate_tree(proc)
            code=proc.wait(timeout=5)
        success=code==0 and (O/'worker_finished.json').exists() and read(O/'worker_finished.json')['success']
    except BaseException:failure=dict(reason='controller_exception',trace=traceback.format_exc())
    finally:
        cleanup=terminate_tree(proc);elapsed=time.perf_counter()-start;meter=read(O/'meter.json')
        pending=max(0,time.perf_counter()-meter['phase_started_perf']) if meter['phase'].startswith('Tab_') and meter['phase_started_perf'] is not None else 0
        tab=min(elapsed,meter['Tab_seconds']+pending);other=elapsed-tab
        good=tab<=LIMITS['Tab_seconds'] and other<=LIMITS['other_seconds'] and elapsed<=LIMITS['stage_seconds'] and peak<=LIMITS['rss_bytes']
        success=bool(success and failure is None and cleanup['reaped'] and good)
        write(O/'run_finished.json',dict(at=now(),success=success,seconds=elapsed,Tab_seconds=tab,other_seconds=other,peak_process_tree_RSS_bytes=peak,
             cap_satisfied=good,failure=failure,returncode=code,worker_reaped=cleanup['reaped'],meter=meter))
        if cleanup['reaped']:settle()
    print(json.dumps(dict(success=success,seconds=elapsed,failure=failure)))
    if not success:raise SystemExit(1)

def settle():

    f,a,l=read(O/'run_finished.json'),read(O/'authorization.json'),read(L)
    assert f['worker_reaped'] and sha(L)==a['ledger_reserved_sha256'] and KEY not in l['new_cheap_diagnostic_seconds']
    m=f['meter'];l['used']['TabICL_contexts']+=m['contexts_attempted'];l['used']['TabICL_query_rows']+=m['query_rows_attempted'];l['used']['TabICL_fit_predict_seconds']+=f['Tab_seconds']
    l['new_cheap_diagnostic_seconds'][KEY]=f['other_seconds']
    l['cheap_stage_details'].append(dict(stage=KEY,seconds=f['other_seconds'],HGB_fits=0,RCTL_fits=0,complete=f['success']))
    l['plan_reservations'][KEY].update(status='settled',Tab_contexts_used=m['contexts_attempted'],Tab_query_rows_used=m['query_rows_attempted'],Tab_seconds_used=f['Tab_seconds'],cheap_seconds_used=f['other_seconds'],complete=f['success'],limit_overrun=not f['cap_satisfied'])
