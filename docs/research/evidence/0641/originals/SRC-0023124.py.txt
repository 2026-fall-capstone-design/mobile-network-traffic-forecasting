def terminate_tree(proc):
    import psutil
    errors=[];children=[]
    if proc is None:return dict(reaped=True,errors=[],ended_perf=time.perf_counter())
    try:
        if proc.poll() is None:
            try:
                parent=psutil.Process(proc.pid);children=parent.children(recursive=True)
            except psutil.NoSuchProcess:pass
            for child in children:
                try:child.kill()
                except psutil.NoSuchProcess:pass
                except Exception as e:errors.append(repr(e))
            try:proc.kill()
            except ProcessLookupError:pass
        proc.wait(timeout=5)
    except Exception as e:errors.append(repr(e))
    live=[]
    if children:
        try:
            _,live=psutil.wait_procs(children,timeout=3)
        except Exception as e:errors.append(repr(e));live=children
    return dict(reaped=proc.poll() is not None and not live,errors=errors,ended_perf=time.perf_counter())

def run():
    import psutil
    a=read(O/'authorization.json');cfg=read(O/'frozen_settings.json')
    assert sha(L)==a['ledger_reserved_sha256']
    assert sha(O/'frozen_settings.json')==a['settings_sha256']
    assert all(sha(B/p)==h for p,h in cfg['source_hashes'].items())
    assert not (O/'run_started.json').exists(),'No duplicate attempt'
    write(O/'meter.json',dict(Tab_seconds=0.,contexts_attempted=0,contexts_completed=0,
        query_rows_attempted=0,query_rows_completed=0,simple_fits_attempted=0,simple_fits_completed=0,
        phase='worker_import',phase_started_perf=None,simple_rows_completed=0))
    start=time.perf_counter();proc=None;peak=0;failure=None;success=False;code=None;tab=0.;meter=read(O/'meter.json');ended_perf=None;stop_requested_perf=None
    write(O/'run_started.json',dict(at=now(),controller_pid=os.getpid(),started_perf=start))
    try:
        with (O/'stdout.log').open('w',encoding='utf-8') as out,(O/'stderr.log').open('w',encoding='utf-8') as err:
            proc=subprocess.Popen([sys.executable,'-X','utf8',str(Path(__file__).resolve()),'worker'],stdout=out,stderr=err,creationflags=getattr(subprocess,'CREATE_NO_WINDOW',0))
            write(O/'worker_handle.json',dict(at=now(),pid=proc.pid,create_time=psutil.Process(proc.pid).create_time()))
            while proc.poll() is None:
                meter=read(O/'meter.json');elapsed=time.perf_counter()-start
                active=(time.perf_counter()-meter['phase_started_perf']) if meter.get('phase_started_perf') is not None and str(meter.get('phase','')).startswith('Tab_') else 0.
                tab=meter.get('Tab_seconds',0.)+active;other=max(0.,elapsed-tab)
                try:
                    ps=psutil.Process(proc.pid);rss=ps.memory_info().rss+sum(c.memory_info().rss for c in ps.children(recursive=True));peak=max(peak,rss)
                except psutil.NoSuchProcess:pass
                if tab>125 or other>25 or elapsed>150 or peak>LIMITS['rss_bytes']:
                    failure=dict(reason='time_or_RSS_cap',Tab_seconds=tab,other_seconds=other,seconds=elapsed,RSS=peak);break
                time.sleep(.05)
            if failure is not None:
                stop_requested_perf=time.perf_counter()
                cleanup=terminate_tree(proc)
                if not cleanup['reaped']:raise RuntimeError('Worker could not be reaped: '+repr(cleanup))
                ended_perf=cleanup['ended_perf']
            code=proc.wait(timeout=5)
            if ended_perf is None:ended_perf=time.perf_counter()
        success=code==0 and (O/'worker_finished.json').exists() and read(O/'worker_finished.json').get('success',False)
    except BaseException:
        stop_requested_perf=time.perf_counter()
        failure=dict(reason='controller_or_worker_observation_exception',trace=traceback.format_exc())
    finally:
        cleanup=terminate_tree(proc)
        if ended_perf is None:ended_perf=cleanup['ended_perf']
        elapsed=time.perf_counter()-start
        meter_fallback=False
        try:meter=read(O/'meter.json')
        except Exception as e:
            meter_fallback=True
            failure=dict(reason='meter_read_failure',error=repr(e),previous_failure=failure)
            # Unobservable last attempt is charged conservatively, not silently omitted.
            meter.update(contexts_attempted=LIMITS['contexts'],query_rows_attempted=LIMITS['query_rows'],simple_fits_attempted=LIMITS['simple_fits'])
        active=0.
        if meter.get('phase_started_perf') is not None and str(meter.get('phase','')).startswith('Tab_'):
            active=max(0.,(stop_requested_perf if stop_requested_perf is not None else ended_perf)-meter['phase_started_perf'])
        tab=min(elapsed,meter.get('Tab_seconds',0.)+active);other=max(0.,elapsed-tab)
        limits_met=tab<=125 and other<=25 and elapsed<=150 and peak<=LIMITS['rss_bytes']
        success=bool(success and limits_met and failure is None and cleanup['reaped'])
        if code is None and proc is not None:code=proc.returncode
        write(O/'run_finished.json',dict(at=now(),success=success,seconds=elapsed,Tab_seconds=tab,other_seconds=other,
            cap_satisfied=limits_met,peak_process_tree_RSS_bytes=peak,failure=failure,returncode=code,
            worker_reaped=cleanup['reaped'],cleanup=cleanup,meter=meter,meter_fallback_conservative_counts=meter_fallback,
            active_Tab_seconds_added_after_forced_stop=active,active_Tab_seconds_are_estimated=active>0,
            cleanup_charged_to_other=True,new_RCTL_fits=0))
        if cleanup['reaped']:settle()
    print(json.dumps(dict(success=success,Tab_seconds=tab,other_seconds=other,seconds=elapsed,failure=failure),ensure_ascii=False))
    if not success:raise SystemExit(1)
