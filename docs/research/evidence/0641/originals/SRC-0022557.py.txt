for k in ('OMP_NUM_THREADS','MKL_NUM_THREADS','OPENBLAS_NUM_THREADS','NUMEXPR_NUM_THREADS'):
    os.environ[k]='1'
import json,hashlib,math,shutil,subprocess,sys,time,traceback
from pathlib import Path
from datetime import datetime,timezone
from recovery_io_271 import write
B=Path(__file__).resolve().parent; ROOT=B.parent.parent
KEY='cached_relation_score_ablation_641'; O=B/'results'/KEY
C=B/'results/bounded_relation_cache_resume_637'; P=B/'results/bounded_relation_precision_638'
S=B/'research_state.json'; L=B/'cumulative_execution_budget.json'; CAP=12.

def reserve():
    s,l=read(S),read(L)
    assert not O.exists(),'Inspect prior execution, do not replay'
    assert s['latest_research_stage']==640 and s['next_action']['id']=='641_cached_relation_score_ablation' and s['in_flight_execution'] is None
    assert sha(L)==read(B/'results/relation_contribution_640/archive_complete.json')['ledger_sha256']
    assert read(P/'verification.json')['success'] and read(P/'settled.json')['success']
    assert all(v['status'] in ('settled','cancelled','completed','released') for fam in ['numeric_reservations','plan_reservations'] for v in l[fam].values())
    assert all(v['status']=='settled' for a in l['additional_rounds'].values() for v in a.get('reservations',{}).values())
    paths=[Path(__file__),B/'641_cached_relation_score_ablation_plan.md',B/'bounded_relation_graph_635.py',B/'recovery_io_271.py',B/'run_joint_ID_decision_558.py',
        C/'frozen_settings.json',P/'graph_scores.npz',P/'assignment_sealed.json',P/'verification.json',P/'settled.json']
    paths+=sorted((C/'cells').glob('c*.npz'))+sorted((P/'edges').glob('*.npz'))
    O.mkdir();(O/'before').mkdir()
    for p in [S,L]:shutil.copy2(p,O/'before'/p.name)
    cfg=read(C/'frozen_settings.json')
    cfg.update(source_hashes={p.relative_to(B).as_posix():sha(p) for p in paths},seconds_cap=CAP,new_models=0,
        controls=['median_distance','plugin_conflict'],models=['Tab','HGB'],read_query_y=False,read_RCTL=False,all_periods_development=True)
    write(O/'frozen_settings.json',cfg)
    before=sha(L);used=l['used'];cheap=math.fsum(l['new_cheap_diagnostic_seconds'].values())
    total=cheap+math.fsum(used.get(k,0.) for k in ['TabICL_fit_predict_seconds','RCTL_wall_seconds','RCTL_frozen_inference_seconds','RCTL_frozen_gradient_seconds'])
    assert l['caps']['cheap_numeric_seconds']>=cheap+CAP and l['caps']['total_local_modeling_seconds']>=total+CAP
    l['numeric_reservations'][KEY]=dict(status='reserved',seconds_cap=CAP,new_models=0,settings_sha256=sha(O/'frozen_settings.json'))
    write(L,l)
    write(O/'authorization.json',dict(at=now(),ledger_before_sha256=before,ledger_reserved_sha256=sha(L),settings_sha256=sha(O/'frozen_settings.json'),cap_changes={}))
    s['in_flight_execution']=dict(id=KEY,output=rel(O));s['next_action'].update(status='reserved',budget_reserved=True);s['updated_at']=now();write(S,s)
    print(json.dumps(dict(reserved=True,new_models=0,seconds_cap=CAP,cap_changes={})))

def worker():
    import numpy as np,psutil
    from bounded_relation_graph_635 import assign
    pp=psutil.Process();pp.cpu_affinity(pp.cpu_affinity()[:4])
    cfg=read(O/'frozen_settings.json');ids=cfg['cell_ids'];checks=[];maxdiff=0.
    def check(name,ok):
        assert bool(ok),name
        checks.append(name)
    with np.load(P/'graph_scores.npz',allow_pickle=False) as z:
        edges=z['edges'].copy();check('same_ids',z['cell_ids'].tolist()==ids);check('initial',z['initial_labels'].tolist()==cfg['initial_labels'])
    check('22edges',edges.shape==(22,2))
    pred={};qt=None
    for p in sorted((C/'cells').glob('c*.npz')):
        i=int(p.stem[1:])
        with np.load(p,allow_pickle=False) as z:
            check('id',int(z['cell_id'])==ids[i]);ts=z['query_times'].copy()
            if qt is None:qt=ts
            check('64samequery',len(ts)==64 and np.array_equal(qt,ts))
            pred[i]=dict(targets=z['target_indices'].tolist(),Tab=z['Tab'].astype(np.float64),HGB=z['HGB'].astype(np.float64))
    check('14cachecells',len(pred)==14)
    costs={f'{model}_{control}':[] for model in cfg['models'] for control in cfg['controls']};records=[]
    for i0,j0 in edges:
        i,j=int(i0),int(j0);saved={}
        for model in cfg['models']:
            # Neither y nor d nor residual score is read from these source archives.
            with np.load(P/'edges'/f'{model}_{i}_{j}.npz',allow_pickle=False) as z:
                saved[model]={k:z[k].copy() for k in ['mask','p','w','F0','F1']}
        mask=saved['Tab']['mask'];check('same_support',np.array_equal(mask,saved['HGB']['mask']))
        check('enough_each',all(mask[o+lo:o+hi].sum()>=8 for o in [0,64] for lo,hi in [(0,32),(32,64)]))
        for model in cfg['models']:
            a,b=pred[i],pred[j]
            v1=np.concatenate([a[model][a['targets'].index(t),:,4] for t in [i,j]])
            v0=np.concatenate([b[model][b['targets'].index(t),:,4] for t in [i,j]])
            z=saved[model];vp=z['w']*z['p']*(1-z['p'])*(z['F0']-z['F1']);vm=np.abs(v1-v0)
            check('finite',np.isfinite(vp).all() and np.isfinite(vm).all())
            check('plugin_nonnegative',np.min(vp)>=-1e-14)
            for control,values in [('median_distance',vm),('plugin_conflict',vp)]:
                vals=[]
                for lo,hi in [(0,32),(32,64)]:
                    vector=math.fsum(float(np.mean(values[o+lo:o+hi][mask[o+lo:o+hi]],dtype=np.float64)) for o in [0,64])/2
                    means=[]
                    for off in [0,64]:
                        row=[]
                        for t in range(off+lo,off+hi):
                            if not mask[t]:continue
                            value=(abs(float(v1[t])-float(v0[t])) if control=='median_distance' else float(z['w'][t])*float(z['p'][t])*(1-float(z['p'][t]))*(float(z['F0'][t])-float(z['F1'][t])))
                            row.append(value)
                        means.append(math.fsum(row)/len(row))
                    scalar=math.fsum(means)/2;diff=abs(vector-scalar);maxdiff=max(maxdiff,diff)
                    check('edge_mean_independent',diff<1e-12);vals.append(max(0.,vector))
                costs[f'{model}_{control}'].append(vals)
        records.append(dict(cells=[ids[i],ids[j]],common_rows=int(mask.sum())))
    assignments={name:assign(ids,cfg['initial_labels'],edges,c,cfg,check) for name,c in costs.items()}
    # Independent list-only graph objective, whole objective recomputed for every proposal.
    for name,c in costs.items():
        lab=list(cfg['initial_labels']);steps=[]
        def J(x):return [math.fsum(c[e][h] for e,(i,j) in enumerate(edges) if x[int(i)]==x[int(j)]) for h in [0,1]]
        for sweep in range(3):
            moved=False
            for i in range(16):
                src=lab[i]
                if lab.count(src)<=1:continue
                before=J(lab);opts=[]
                for dst in range(4):
                    if dst==src:continue
                    neighbors=[int(j if int(a)==i else a) for a,j in edges if int(a)==i or int(j)==i]
                    if sum(lab[j]==dst for j in neighbors)<2:continue
                    q=list(lab);q[i]=dst;after=J(q);delta=[after[h]-before[h] for h in [0,1]]
                    if all(d<-1e-12 for d in delta):opts.append((math.fsum(delta)/2,dst))
                if opts:
                    dst=min(opts)[1];lab[i]=dst;moved=True;steps.append((sweep,ids[i],src,dst))
            if not moved:break
        check('scalar_assignment',lab==assignments[name]['labels'])
        check('scalar_moves',steps==[(v['sweep'],v['cell_id'],v['source'],v['destination']) for v in assignments[name]['moves']])
    original=read(P/'assignment_sealed.json')['assignments']
    canon=lambda a:sorted(tuple(g) for g in a['groups'])
    equality={name:canon(a)==canon(original[name.split('_',1)[0]]) for name,a in assignments.items()}
    count=sum(equality[name] for name in ['Tab_median_distance','Tab_plugin_conflict'])
    branch={2:'both_Tab_controls_same_assignment',1:'one_Tab_control_same_assignment',0:'both_Tab_controls_different_assignment'}[count]
    check('source_bytes_unchanged',all(sha(B/p)==h for p,h in cfg['source_hashes'].items()))
    write(O/'assignment_sealed.json',dict(at=now(),branch=branch,assignments=assignments,original_assignments=original,
        same_as_original=equality,RCTL_read=False,query_y_read=False,new_models=0,all_periods_development=True))
    with (O/'graph_scores.npz').open('wb') as f:np.savez_compressed(f,edges=edges,**{k:np.asarray(v) for k,v in costs.items()})
    write(O/'verification.json',dict(at=now(),success=True,checks=len(checks),max_scalar_difference=maxdiff,source_bytes_unchanged=True,RCTL_read=False,query_y_read=False,new_models=0))
    write(O/'result.json',dict(at=now(),success=True,branch=branch,assignments=assignments,same_as_original=equality,edges=22,cache_cells=14,
        edge_records=records,new_models=0,new_forecasts=0,RCTL_read=False,query_y_read=False,all_periods_development=True))

def run():
    import psutil
    from run_joint_ID_decision_558 import terminate_tree
    a=read(O/'authorization.json');cfg=read(O/'frozen_settings.json')
    assert sha(L)==a['ledger_reserved_sha256'] and sha(O/'frozen_settings.json')==a['settings_sha256'] and not (O/'run_started.json').exists()
    assert all(sha(B/p)==h for p,h in cfg['source_hashes'].items())
    start=time.perf_counter();proc=None;peak=0;failure=None;success=False
    write(O/'run_started.json',dict(at=now(),controller_pid=os.getpid()))
    try:
        with (O/'stdout.log').open('w') as out,(O/'stderr.log').open('w') as err:
            proc=subprocess.Popen([sys.executable,'-X','utf8',str(Path(__file__).resolve()),'worker'],stdout=out,stderr=err,creationflags=getattr(subprocess,'CREATE_NO_WINDOW',0))
            while proc.poll() is None:
                try:peak=max(peak,psutil.Process(proc.pid).memory_info().rss)
                except psutil.NoSuchProcess:pass
                if time.perf_counter()-start>CAP-.3 or peak>1024**3:failure='resource_cap';break
                time.sleep(.05)
            if failure:terminate_tree(proc)
            code=proc.wait(timeout=3)
        success=code==0 and (O/'result.json').exists() and read(O/'result.json')['success']
    except BaseException:failure=traceback.format_exc()
    finally:
        cleanup=terminate_tree(proc);seconds=time.perf_counter()-start;success=bool(success and not failure and seconds<=CAP and cleanup['reaped'])
        write(O/'run_finished.json',dict(at=now(),success=success,seconds=seconds,failure=failure,worker_reaped=cleanup['reaped'],peak_RSS=peak,new_models=0))
        if cleanup['reaped']:
            l=read(L);assert sha(L)==a['ledger_reserved_sha256'];l['new_cheap_diagnostic_seconds'][KEY]=seconds
            l['cheap_stage_details'].append(dict(stage=KEY,seconds=seconds,new_models=0,complete=success))
            l['numeric_reservations'][KEY].update(status='settled',seconds_used=seconds,complete=success,limit_overrun=seconds>CAP);l['as_of_date']=now();write(L,l)
            s=read(S);assert s['in_flight_execution']['id']==KEY;s['in_flight_execution']=None;s['next_action'].update(status='executed_settled_pending_findings',budget_reserved=False);s['updated_at']=now();write(S,s)
            write(O/'settled.json',dict(at=now(),success=success,seconds=seconds,new_models=0,ledger_sha256=sha(L),cap_changes={},all_prior_costs_preserved=True))
    print(json.dumps(dict(success=success,seconds=seconds,failure=failure)))
    if not success:raise SystemExit(1)

if __name__=='__main__': {'reserve':reserve,'worker':worker,'run':run}[sys.argv[1]]()
