def worker():
    import numpy as np,psutil
    pp=psutil.Process();pp.cpu_affinity(pp.cpu_affinity()[:4]);cfg=read(O/'frozen_settings.json');ids=cfg['cell_ids'];checks=[];maxroot=0.
    def check(name,ok):
        assert bool(ok),name
        checks.append(name)
    with np.load(B/'results/fixed_reference_vs_joint_pool_616/errors_0.npz',allow_pickle=False) as z:truth=z['truth'].copy();qt=z['query_times'].copy();check('truth_ids',z['cell_ids'].tolist()==ids)
    pred={};dtypes={}
    for p in sorted((P/'cells').glob('c*.npz')):
        i=int(p.stem[1:])
        with np.load(p,allow_pickle=False) as z:
            check('cell_id',int(z['cell_id'])==ids[i]);check('cell_times',np.array_equal(qt,z['query_times']))
            pred[i]=dict(targets=z['target_indices'].tolist(),Tab=z['Tab'].astype(np.float64),HGB=z['HGB'].astype(np.float64))
            dtypes[str(i)]=dict(Tab=str(z['Tab'].dtype),HGB=str(z['HGB'].dtype))
    def enough(mask):
        return all(mask[off+lo:off+hi].sum()>=8 and len(np.unique(qt[lo:hi][mask[off+lo:off+hi]]//24))>=2 for off in [0,64] for lo,hi in [(0,32),(32,64)])
    def oracle(k1,k0,p):
        knots=np.unique(np.r_[k1,k0]);f=p*np.interp(knots,k1,np.arange(1,10)/10,left=.1,right=.9)+(1-p)*np.interp(knots,k0,np.arange(1,10)/10,left=.1,right=.9)
        h=int(np.searchsorted(f,.5,side='left'))
        if h==0:return float(knots[0])
        check('oracle_crossing',h<len(knots) and f[h]>f[h-1])
        return float(knots[h-1]+(.5-f[h-1])/(f[h]-f[h-1])*(knots[h]-knots[h-1]))
    records=[];edges=[];cost={'Tab':[],'HGB':[]};recovered={'Tab':0,'HGB':0}
    for rec in read(C/'retrieval_sealed.json')['records']:
        if not rec['retained']:continue
        i,j=rec['i'],rec['j'];y=np.r_[truth[i],truth[j]];d=np.r_[np.ones(64),np.zeros(64)]
        with np.load(C/'edges'/f'p_{i}_{j}.npz',allow_pickle=False) as z:p=z['p'].copy();support=z['mask'].copy()
        allr={};oldr={}
        for model in ['Tab','HGB']:
            a,b=pred[i],pred[j];k1=np.concatenate([a[model][a['targets'].index(t)] for t in [i,j]]);k0=np.concatenate([b[model][b['targets'].index(t)] for t in [i,j]])
            r=m.relationship(k1,k0,p,y,d);allr[model]=r
            with np.load(P/'edges'/f'{model}_{i}_{j}.npz',allow_pickle=False) as z:oldr[model]={k:z[k].copy() for k in ['valid','mixture','q']}
            recovered[model]+=int(np.sum(r['valid']&~oldr[model]['valid']))
            for t in np.flatnonzero(r['valid']):
                root=oracle(k1[t],k0[t],float(p[t]));diff=abs(root-float(r['q'][t]));maxroot=max(maxroot,diff)
                check('independent_linear_root',diff<1e-8)
                q=float(r['q'][t]);f1=float(np.interp(q,k1[t],np.arange(1,10)/10,left=.1,right=.9));f0=float(np.interp(q,k0[t],np.arange(1,10)/10,left=.1,right=.9))
                check('independent_CDF',max(abs(f1-r['F1'][t]),abs(f0-r['F0'][t]))<1e-10)
                check('bounded_score',abs(float(r['score'][t]))<=.5+1e-12)
        mask=support&allr['Tab']['valid']&allr['HGB']['valid'];ok=enough(mask);stats={}
        for model,r in allr.items():
            vals=[]
            if ok:
                for lo,hi in [(0,32),(32,64)]:
                    av=[]
                    for off in [0,64]:
                        ix=[t for t in range(off+lo,off+hi) if mask[t]]
                        score=[float(r['w'][t])*(float(d[t])-float(r['a'][t]))*(.5-int(float(y[t])<=float(r['q'][t]))) for t in ix]
                        av.append(math.fsum(score)/len(score))
                    vals.append(max(0.,math.fsum(av)/2))
                cost[model].append(vals)
            stats[model]=dict(valid_rows=int(r['valid'].sum()),old_valid_rows=int(oldr[model]['valid'].sum()),cost=vals)
            tmp=O/'edges'/f'{model}_{i}_{j}.npz'
            with tmp.open('wb') as f:np.savez_compressed(f,**r,p=p,mask=mask,y=y,d=d)
        if ok:edges.append((i,j))
        records.append(dict(cell_ids=[ids[i],ids[j]],common_supported=ok,common_rows=int(mask.sum()),stats=stats))
    assignments={name:m.assign(ids,cfg['initial_labels'],edges,cost[name],cfg,check) for name in ['Tab','HGB']}
    same=sorted(assignments['Tab']['groups'])==sorted(assignments['HGB']['groups'])
    branch=('no_common_supported_edges' if not edges else 'Tab_keeps_UPC' if not assignments['Tab']['changed_cell_ids'] else 'same_changed_assignment' if same else 'distinct_Tab_assignment')
    check('source_unchanged',all(sha(B/p)==h for p,h in cfg['source_hashes'].items()))
    write(O/'assignment_sealed.json',dict(at=now(),branch=branch,assignments=assignments,RCTL_read=False,all_periods_development=True))
    with (O/'graph_scores.npz').open('wb') as f:np.savez_compressed(f,edges=np.asarray(edges,int).reshape(-1,2),Tab=np.asarray(cost['Tab']).reshape(-1,2),HGB=np.asarray(cost['HGB']).reshape(-1,2),cell_ids=ids,initial_labels=cfg['initial_labels'])
    write(O/'verification.json',dict(at=now(),success=True,checks=len(checks),maximum_independent_root_difference=maxroot,source_hashes_unchanged=True,new_model_runs=0))
    write(O/'result.json',dict(at=now(),success=True,branch=branch,common_valid_edges=len(edges),previous_edges=2,recovered_rows=recovered,original_dtypes=dtypes,
        assignments=assignments,edge_records=records,new_models=0,RCTL_read=False,all_periods_development=True))

def run():
    import psutil
    from run_joint_ID_decision_558 import terminate_tree
    a=read(O/'authorization.json');cfg=read(O/'frozen_settings.json')
    assert sha(L)==a['ledger_reserved_sha256'] and sha(O/'frozen_settings.json')==a['settings_sha256'] and not (O/'run_started.json').exists()
    assert all(sha(B/p)==h for p,h in cfg['source_hashes'].items())
    start=time.perf_counter();proc=None;peak=0;failure=None;success=False
    write(O/'run_started.json',dict(at=now(),controller_pid=os.getpid()))
    try:
        with (O/'stdout.log').open('w') as out,(O/'stderr.log').open('w') as err:
            proc=subprocess.Popen([sys.executable,'-X','utf8',str(Path(__file__).resolve()),'worker'],stdout=out,stderr=err,creationflags=getattr(subprocess,'CREATE_NO_WINDOW',0))
            while proc.poll() is None:
                try:peak=max(peak,psutil.Process(proc.pid).memory_info().rss)
                except psutil.NoSuchProcess:pass
                if time.perf_counter()-start>CAP-.3 or peak>2*1024**3:failure='resource_cap';break
                time.sleep(.05)
            if failure:terminate_tree(proc)
            code=proc.wait(timeout=3)
        success=code==0 and (O/'result.json').exists() and read(O/'result.json')['success']
    except BaseException:failure=traceback.format_exc()
    finally:
        cleanup=terminate_tree(proc);seconds=time.perf_counter()-start;success=bool(success and not failure and seconds<=CAP and cleanup['reaped'])
        f=dict(at=now(),success=success,seconds=seconds,failure=failure,worker_reaped=cleanup['reaped'],peak_RSS=peak,new_models=0);write(O/'run_finished.json',f)
        if cleanup['reaped']:
            l=read(L);assert sha(L)==a['ledger_reserved_sha256'];l['new_cheap_diagnostic_seconds'][KEY]=seconds
            l['cheap_stage_details'].append(dict(stage=KEY,seconds=seconds,new_models=0,complete=success))
            l['numeric_reservations'][KEY].update(status='settled',seconds_used=seconds,complete=success,limit_overrun=seconds>CAP);write(L,l)
            s=read(S);assert s['in_flight_execution']['id']==KEY;s['in_flight_execution']=None;s['next_action'].update(status='executed_settled_pending_findings',budget_reserved=False);s['updated_at']=now();write(S,s)
            write(O/'settled.json',dict(at=now(),success=success,seconds=seconds,new_models=0,ledger_sha256=sha(L),cap_changes={},all_prior_costs_preserved=True))
    print(json.dumps(dict(success=success,seconds=seconds,failure=failure)))
    if not success:raise SystemExit(1)

if __name__=='__main__':
    if sys.argv[1]=='worker':worker()
    else:{'reserve':reserve,'run':run}[sys.argv[1]]()
