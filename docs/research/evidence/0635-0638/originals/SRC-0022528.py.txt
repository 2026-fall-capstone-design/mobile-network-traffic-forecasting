def cdf_pdf(knots,y):
    import numpy as np
    n=len(y);pos=np.sum(y[:,None]>=knots,axis=1)-1;ix=np.clip(pos,0,7);row=np.arange(n)
    gaps=knots[row,ix+1]-knots[row,ix]
    safe=np.maximum(gaps,1e-300)
    F=(ix+1)/10+.1*(y-knots[row,ix])/safe
    f=.1/safe
    lower=y<knots[:,0];upper=y>=knots[:,-1]
    F=np.where(lower,.1,np.where(upper,.9,F));f=np.where(lower|upper,0.,f)
    return F,f

def relationship(k1,k0,p,y,d):
    import numpy as np
    span=np.maximum(1.,np.maximum(np.max(np.abs(k1),axis=1),np.max(np.abs(k0),axis=1)))
    valid=np.all(np.diff(k1,axis=1)>1e-10*span[:,None],axis=1)&np.all(np.diff(k0,axis=1)>1e-10*span[:,None],axis=1)
    lo=np.minimum(k1[:,0],k0[:,0]).copy();hi=np.maximum(k1[:,-1],k0[:,-1]).copy()
    # Equal operations for the matched estimators. No query labels in q/a/w.
    for _ in range(48):
        mid=(lo+hi)/2;F1,_=cdf_pdf(k1,mid);F0,_=cdf_pdf(k0,mid);left=(p*F1+(1-p)*F0)<.5
        lo=np.where(left,mid,lo);hi=np.where(left,hi,mid)
    q=(lo+hi)/2;F1,f1=cdf_pdf(k1,q);F0,f0=cdf_pdf(k0,q);fm=p*f1+(1-p)*f0
    valid &= np.isfinite(fm)&(fm>0)&(np.abs(p*F1+(1-p)*F0-.5)<1e-7)
    a=np.divide(p*f1,fm,out=np.zeros_like(p),where=fm>0);w=np.sign(F0-F1)
    score=w*(d-a)*(.5-(y<=q))
    return dict(q=q,a=a,w=w,score=score,valid=valid,F1=F1,F0=F0,f1=f1,f0=f0,mixture=p*F1+(1-p)*F0)

def assign(ids,initial,edges,costs,cfg,check):
    import numpy as np
    lab=np.asarray(initial,int).copy();e=np.asarray(edges,int).reshape(-1,2);c=np.asarray(costs,float).reshape(-1,2)
    def J(labels):return np.sum(c[labels[e[:,0]]==labels[e[:,1]]],axis=0)
    steps=[];base=J(lab); stop='three_sweeps'
    for sweep in range(cfg['sweeps']):
        moved=0
        for i in range(len(ids)):
            src=int(lab[i])
            if np.sum(lab==src)<=1:continue
            choices=[];before=J(lab)
            for dest in range(4):
                if dest==src:continue
                neighbors=[int(b if a==i else a) for a,b in e if a==i or b==i]
                if sum(lab[j]==dest for j in neighbors)<2:continue
                proposal=lab.copy();proposal[i]=dest;delta=J(proposal)-before
                # Independent scalar sum over only incident edges.
                ref=[]
                for h in range(2):
                    ref.append(math.fsum(float(c[k,h])*(int(proposal[a]==proposal[b])-int(lab[a]==lab[b])) for k,(a,b) in enumerate(e) if a==i or b==i))
                check('move_delta_scalar',np.max(np.abs(delta-np.asarray(ref)))<1e-12)
                if np.all(delta < -1e-12):choices.append((float(delta.mean()),dest,delta))
            if not choices:continue
            _,dest,delta=min(choices,key=lambda z:(z[0],z[1]));lab[i]=dest;moved+=1
            check('nonempty_K',set(lab.tolist())==set(range(4)))
            check('both_decrease',np.all(J(lab)<before-1e-12))
            steps.append(dict(sweep=sweep,cell_id=ids[i],source=src,destination=dest,change=delta.tolist(),objective_after=J(lab).tolist()))
        if moved==0:stop='no_admissible_move';break
    return dict(labels=lab.tolist(),groups=[sorted(ids[i] for i in range(len(ids)) if lab[i]==g) for g in range(4)],
        changed_cell_ids=[ids[i] for i in range(len(ids)) if lab[i]!=initial[i]],initial_objective=base.tolist(),final_objective=J(lab).tolist(),moves=steps,stop=stop)

        success=code==0 and (O/'worker_finished.json').exists() and read(O/'worker_finished.json')['success']
    except BaseException:failure=dict(reason='controller_exception',trace=traceback.format_exc())
    finally:
        cleanup=terminate_tree(proc);elapsed=time.perf_counter()-start;meter=read(O/'meter.json')
        pending=max(0,time.perf_counter()-meter['phase_started_perf']) if meter['phase'].startswith('Tab_') and meter['phase_started_perf'] is not None else 0
        tab=min(elapsed,meter['Tab_seconds']+pending);other=elapsed-tab
        good=tab<=LIMITS['Tab_seconds'] and other<=LIMITS['other_seconds'] and elapsed<=LIMITS['stage_seconds'] and peak<=LIMITS['rss_bytes']
        success=bool(success and failure is None and cleanup['reaped'] and good)
        write(O/'run_finished.json',dict(at=now(),success=success,seconds=elapsed,Tab_seconds=tab,other_seconds=other,peak_process_tree_RSS_bytes=peak,cap_satisfied=good,
            failure=failure,returncode=code,worker_reaped=cleanup['reaped'],meter=meter))
        if cleanup['reaped']:settle()
    print(json.dumps(dict(success=success,seconds=elapsed,failure=failure)))
    if not success:raise SystemExit(1)

def settle():
    assert not (O/'settled.json').exists()
    f,a,l=read(O/'run_finished.json'),read(O/'authorization.json'),read(L)
    assert f['worker_reaped'] and sha(L)==a['ledger_reserved_sha256'] and KEY not in l['new_cheap_diagnostic_seconds']
    m=f['meter'];l['used']['TabICL_contexts']+=m['contexts_attempted'];l['used']['TabICL_query_rows']+=m['query_rows_attempted'];l['used']['TabICL_fit_predict_seconds']+=f['Tab_seconds']

    l['cheap_stage_details'].append(dict(stage=KEY,seconds=f['other_seconds'],HGB_fits=m['quantile_attempted']+m['propensity_attempted'],RCTL_fits=0,complete=f['success']))

        cheap_seconds_used=f['other_seconds'],complete=f['success'],limit_overrun=not f['cap_satisfied'])
    for k in ['TabICL_contexts','TabICL_query_rows']:l['remaining_count_budgets'][k]=l['caps'][k]-l['used'][k]
    l['as_of_date']=now();write(L,l)
    s=read(S);assert s['in_flight_execution']['id']==KEY;s['in_flight_execution']=None;s['next_action'].update(status='executed_settled_pending_findings',budget_reserved=False);s['updated_at']=now();write(S,s)
    write(O/'settled.json',dict(at=now(),success=f['success'],seconds=f['seconds'],Tab_seconds=f['Tab_seconds'],other_seconds=f['other_seconds'],
        new_Tab_contexts=m['contexts_attempted'],new_Tab_query_rows=m['query_rows_attempted'],HGB_new_fits=m['quantile_attempted']+m['propensity_attempted'],RCTL_new_fits=0,
        ledger_sha256=sha(L),all_prior_costs_preserved=True,cap_changes=a['cap_changes']))
