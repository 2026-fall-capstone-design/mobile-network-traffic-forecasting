"""Activate the one authorized post-106 round once; account every adaptation."""
import hashlib
import json
from pathlib import Path
import sys

BASE = Path(__file__).resolve().parent
PATH = BASE / "cumulative_execution_budget.json"
ROUND = "post_revision_106_round1"
OUT = BASE / "results/normalization_108"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write(path, data):
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2, allow_nan=False), encoding="utf-8")


def main(mode):
    ledger = json.loads(PATH.read_text(encoding="utf-8"))
    if mode == "reserve":
        assert not (OUT / "run_started.json").exists()
        assert ROUND not in ledger.get("additional_rounds", {})
        assert ledger["used"]["RCTL_fits"] == 29
        assert ledger["used"]["TabICL_contexts"] == 78
        assert ledger["used"]["TabICL_query_rows"] == 55488
        plan = BASE / "108_normalization_diagnostic_plan.md"
        code = BASE / "normalization_diagnostic_108.py"
        assert plan.exists() and code.exists()
        backup = BASE / "108_ledger_before_round_activation.json"
        assert not backup.exists()
        backup.write_bytes(PATH.read_bytes())
        r107 = json.loads((BASE / "results/support_composition_107/result.json").read_text(encoding="utf-8"))
        assert "support_composition_107" not in ledger["new_cheap_diagnostic_seconds"]
        cheap = r107["arithmetic_seconds"] + r107["prior_failed_process_seconds"]
        ledger["new_cheap_diagnostic_seconds"]["support_composition_107"] = cheap
        ledger["cheap_stage_details"].append({"stage": "support_composition_107", "new_simple_fits": 0,
            "new_RCTL_fits": 0, "new_TabICL_calls": 0, "new_model_forwards": 0,
            "entire_charged_seconds": cheap, "failed_process_seconds_included": r107["prior_failed_process_seconds"],
            "arithmetic_seconds": r107["arithmetic_seconds"], "cost_not_counted_twice": True})
        ledger["used"]["RCTL_statistics_adaptations"] = 0
        ledger["used"]["RCTL_fit_or_adaptation_runs"] = 29
        ledger["used"]["RCTL_statistics_adaptation_stage_seconds"] = 0.0
        ledger["caps"]["RCTL_fits"] = 47
        ledger["caps"]["RCTL_fit_or_adaptation_runs"] = 47
        ledger["caps"]["RCTL_epochs"] = ledger["used"]["RCTL_epochs"] + 18 * 80
        ledger["caps"]["RCTL_wall_seconds"] = ledger["used"]["RCTL_wall_seconds"] + 1800
        ledger["additional_rounds"] = {ROUND: {
            "status": "active", "authorization": "User continuation request reflected in goal section 9.1 on 2026-09-26",
            "plan_sha256": sha(plan), "code_sha256": sha(code),
            "previous_ledger_sha256": sha(backup), "baseline_optimizer_fits": 29,
            "baseline_RCTL_wall_seconds": ledger["used"]["RCTL_wall_seconds"],
            "limit_fit_or_adaptation_runs": 18, "limit_epochs_per_optimizer_fit": 80,
            "limit_RCTL_model_seconds": 1800, "used_fit_or_adaptation_runs": 0,
            "used_optimizer_fits": 0, "used_statistics_adaptations": 0, "used_RCTL_model_seconds": 0.0,
            "TabICL_allowance_activated": False,
            "reservations": {"normalization_108": {"status": "reserved", "adaptations": 9,
                "optimizer_fits": 0, "stage_seconds_cap": 300}},
            "unreserved_remaining_fit_or_adaptation_runs": 9,
            "automatic_recharge": False}}
        ledger["remaining_count_budgets"]["RCTL_fits"] = 9
        ledger["remaining_count_budgets"]["RCTL_fit_or_adaptation_runs_including_reservation"] = 18
        ledger["new_RCTL_fits_permitted_under_current_cap"] = True
        ledger["RCTL_count_accounting_note"] = (
            "used.RCTL_fits counts optimizer training only. Statistics adaptations consume the same combined allowance; "
            "caps.RCTL_fits is a compatibility alias of the combined cap, never subtract only optimizer fits. "
            "remaining_count_budgets.RCTL_fits excludes outstanding reservations. "
            "RCTL_wall_seconds includes completed adaptation stages; the separate adaptation seconds are a subset, not an extra charge.")
        write(PATH, ledger)
        print(json.dumps({"round_activated": ROUND, "reserved_adaptations": 9, "unreserved_runs": 9,
                          "new_stage107_cheap_seconds": cheap, "ledger_sha256": sha(PATH)}))
    elif mode == "settle":
        assert (OUT / "run_finished.json").exists()
        budget = ledger["additional_rounds"][ROUND]
        reservation = budget["reservations"]["normalization_108"]
        assert reservation["status"] == "reserved", "Already accounted; do not charge twice"
        result_path = OUT / ("result.json" if (OUT / "result.json").exists() else "failure.json")
        result = json.loads(result_path.read_text(encoding="utf-8"))
        used = result["counts"]["statistics_adaptations_started"]
        seconds = result["entire_stage_seconds"]
        assert used <= 9 and seconds >= 0
        assert budget["used_fit_or_adaptation_runs"] + used <= 18
        reservation.update({"status": "settled", "used_adaptations": used,
                            "used_stage_seconds": seconds, "result_sha256": sha(result_path),
                            "complete": result.get("complete", False)})
        budget["used_fit_or_adaptation_runs"] += used
        budget["used_statistics_adaptations"] += used
        budget["used_RCTL_model_seconds"] += seconds
        budget["unreserved_remaining_fit_or_adaptation_runs"] = 18 - budget["used_fit_or_adaptation_runs"]
        ledger["used"]["RCTL_statistics_adaptations"] += used
        ledger["used"]["RCTL_fit_or_adaptation_runs"] += used
        ledger["used"]["RCTL_statistics_adaptation_stage_seconds"] += seconds
        ledger["used"]["RCTL_wall_seconds"] += seconds
        ledger["remaining_count_budgets"]["RCTL_fits"] = budget["unreserved_remaining_fit_or_adaptation_runs"]
        ledger["remaining_count_budgets"]["RCTL_fit_or_adaptation_runs_including_reservation"] = 18 - budget["used_fit_or_adaptation_runs"]
        ledger.setdefault("statistics_adaptation_stages", []).append({"stage": "normalization_108",
            "optimizer_fits": 0, "adaptations_started": used,
            "adaptations_completed": result["counts"]["statistics_adaptations_completed"],
            "entire_stage_seconds": seconds, "included_in_RCTL_wall_seconds": True,
            "counts": result["counts"], "complete": result.get("complete", False)})
        ledger["new_RCTL_fits_permitted_under_current_cap"] = budget["unreserved_remaining_fit_or_adaptation_runs"] > 0
        write(PATH, ledger)
        print(json.dumps({"stage": "normalization_108", "charged_adaptations": used,
                          "charged_seconds": seconds, "remaining_round_runs": budget["unreserved_remaining_fit_or_adaptation_runs"],
                          "ledger_sha256": sha(PATH)}))
    else:
        raise ValueError(mode)


if __name__ == "__main__":
    main(sys.argv[1])
