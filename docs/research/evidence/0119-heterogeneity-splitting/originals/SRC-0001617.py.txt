from pathlib import Path
import json
import subprocess

dest = Path(__file__).resolve().parent / 'sources/heterogeneity_89'
render = dest/'rendered'
render.mkdir(exist_ok=True)
poppler = Path('C:/Users/jungj/.cache/codex-runtimes/codex-primary-runtime/dependencies/native/poppler/Library/bin/pdftoppm.exe')
jobs = [
    ('Global_Models_Simulation_2012.12485v3', 19),
    ('Global_Models_Simulation_2012.12485v3', 28),
    ('Global_Models_Simulation_2012.12485v3', 29),
    ('Similarity_Diversity_2404.06198v2', 18),
    ('Similarity_Diversity_2404.06198v2', 25),
    ('Similarity_Diversity_2404.06198v2', 28),
]
records=[]
for stem,p in jobs:
    output=render/f'{stem}_p{p:02d}'
    image_path=Path(str(output)+'.png')
    if image_path.exists(): raise RuntimeError('Render output already exists')
    run=subprocess.run([str(poppler), '-f',str(p),'-l',str(p),'-singlefile','-scale-to','1400','-png',str(dest/(stem+'.pdf')),str(output)],capture_output=True,text=True,check=True)
    records.append({'source':stem+'.pdf','page':p,'image':str(image_path)})
(dest/'render_manifest.json').write_text(json.dumps(records,ensure_ascii=False,indent=2),encoding='utf-8')
print(json.dumps(records,ensure_ascii=True))
