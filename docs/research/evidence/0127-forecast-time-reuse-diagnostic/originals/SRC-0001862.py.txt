def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def read(path):
    return json.loads(path.read_text(encoding='utf-8'))

def save(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False) + '\n', encoding='utf-8')

        assert abs(np.mean(row['cell_mae']) - row['mean_scaled_mae']) < 1e-12
        begin, end = settings['periods'][row['period']]
        refresh = np.arange(begin, end) % row['cadence_hours'] == 0
        assert row['fresh_prediction_rows_arithmetic'] == int(16 * refresh.sum())
        assert row['model_calls_arithmetic'] == int(row['models'] * refresh.sum())
    for method in methods:
        fresh = original[method].astype(np.float64)
        for mode in settings['modes']:
            assert np.allclose(saved[f'{method}__{mode}__d1'], fresh, rtol=0, atol=1e-15)
            for cadence in settings['cadences']:
                take = np.arange(480) % cadence == 0
                assert np.allclose(saved[f'{method}__{mode}__d{cadence}'][:, take], fresh[:, take], rtol=0, atol=1e-15)

def metric(method, cadence=1, period='all', mode='hold_increment'):
    return next(r for r in metrics if (r['method'], r['mode'], r['cadence_hours'], r['period']) == (method, mode, cadence, period))
